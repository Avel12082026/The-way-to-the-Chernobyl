function placement(layers,side,offset=0){
 if(!layers||!['player','enemy'].includes(side))return null;
 const scale=800/layers.body.height,sign=side==='enemy'?-1:1;
 const ground=Number.isFinite(layers.feet?.groundY)?layers.feet.groundY:layers.body.height;
 return {scale,sign,x:(side==='player'?340:1196)+offset,y:940,ground};
}
function project(layers,side,point,offset=0){
 const p=placement(layers,side,offset);if(!p||!Array.isArray(point))return null;
 const {character,weapon,adjustment}=layers,s=weapon.scale*((adjustment?.size??100)/100),angle=(adjustment?.angle||0)*Math.PI/180;
 const dx=(point[0]-weapon.grip[0])*s,dy=(point[1]-weapon.grip[1])*s;
 const x=character.grip[0]+(adjustment?.dx||0)+dx*Math.cos(angle)-dy*Math.sin(angle);
 const y=character.grip[1]+(adjustment?.dy||0)+dx*Math.sin(angle)+dy*Math.cos(angle);
 return {x:p.x+p.sign*p.scale*(x-layers.body.width/2),y:p.y+p.scale*(y-p.ground)};
}
function muzzle(layers,side,offset=0){
 const source=layers?.muzzle;if(!source||!Array.isArray(source.muzzle)||!Array.isArray(source.direction))return null;
 const tip=project(layers,side,source.muzzle,offset),ahead=project(layers,side,[source.muzzle[0]+source.direction[0],source.muzzle[1]+source.direction[1]],offset);
 if(!tip||!ahead)return null;
 return {...tip,angle:Math.atan2(ahead.y-tip.y,ahead.x-tip.x),scale:placement(layers,side,offset).scale,kind:source.kind,suppressed:!!source.suppressed,weaponId:layers.weaponId};
}
function feet(layers,side,offset=0){
 const p=placement(layers,side,offset);if(!p||!Array.isArray(layers.feet?.feet))return [];
 return layers.feet.feet.map(foot=>({x:p.x+p.sign*p.scale*(foot.x-layers.body.width/2),y:p.y+p.scale*(foot.y-p.ground),width:foot.width*p.scale}));
}
